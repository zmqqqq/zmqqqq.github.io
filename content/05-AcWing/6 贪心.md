## 区间覆盖

![[Pasted image 20260509200053.png]]

![[Pasted image 20260509200112.png]]

![[Pasted image 20260509202743.png]]

```C++
#include<bits/stdc++.h>
using namespace std;

const int N = 100010;

struct Range{
    int l, r;
};

bool cmp(Range a, Range b){
    return a.l < b.l; // 按左端点从小到大排序
}

int main(){
    Range range[N];

    int st, ed;
    cin >> st >> ed;

    int n;
    cin >> n;

    for(int i = 0; i < n; i++){
        cin >> range[i].l >> range[i].r;
    }

    sort(range, range + n, cmp);

    int res = 0;
    bool success = false;

    for(int i = 0; i < n; i++){
        int j = i;
        int r = -2e9;

        // 找所有左端点 <= st 的区间
        // 在这些区间里，选择右端点最大的那个
        while(j < n && range[j].l <= st){
            r = max(r, range[j].r);
            j++;
        }

        // 如果这些区间都不能覆盖到 st，说明无解
        if(r < st){
            break;
        }

        res++;

        // 如果已经覆盖到目标右端点 ed，成功
        if(r >= ed){
            success = true;
            break;
        }

        // 更新当前已经覆盖到的位置
        st = r;

        // 下一轮从 j 开始
        i = j - 1;
    }

    if(success) cout << res << endl;
    else cout << -1 << endl;

    return 0;
}
```
