## 1 DFS

有两个很重要的概念：==回溯==与==剪枝==。

DFS问题没有模板，重要的是==思路==。

### 排列数字（全排列）

==DFS+回溯==

![[Pasted image 20260402194119.png|395]]

![[Pasted image 20260402195412.png]]

![[Pasted image 20260402203715.png]]

```c++
#include<bits/stdc++.h>
using namespace std;

const int N = 10;

int n;
int path[N]; // 存状态
bool st[N]; // 表示true的时候代表这个点被用过了

void dfs(int u){
    if(u==n){
        for(int i=0;i<n;i++) cout<<path[i]<<" ";
        cout<<endl;
        return;
    }
    // else u<n的情况
    for(int i=1;i<=n;i++){
        if(!st[i]){ // 如果st[i]没有被用过
            path[u]=i;
            st[i]=true;
            dfs(u+1);
            // 恢复现场
            path[u]=0;
            st[i]=false;
        }
    }
}

int main(){
    cin>>n;
    dfs(0);
}
```

### n-皇后问题

![[Pasted image 20260403094824.png]]
![[Pasted image 20260403094848.png|299]]

```c++
#include<bits/stdc++.h>
using namespace std;

const int N = 20;

int n;
char g[N][N]; // 一组方案
bool col[N], dg[N], udg[N]; // 正对角线与反对角线

void dfs(int u){
    if(u==n){
        for(int i=0;i<n;i++) cout<<g[i]<<endl;
        cout<<endl;
        return;
    }
    // else u<n的情况
    for(int i=0;i<n;i++){
        if(!col[i] && !dg[u+i] && !udg[n-u+i]){
            g[u][i]='Q';
            col[i]=dg[u+i]=udg[n-u+i]=true;
            dfs(u+1);
            // 恢复现场
            col[i]=dg[u+i]=udg[n-u+i]=false;
            g[u][i]='.';
        }
    }
}

int main(){
    cin>>n;
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            g[i][j]='.';
        }
    }
    dfs(0);
}
```

### 走迷宫

![[Pasted image 20260403123017.png]]

从起始位置开始搜索，第一步将距离为1的点放进来，第二步所有距离为2的点...

==每一个数字表示它是第几层被扩展到的：==

![[Pasted image 20260403123434.png|249]]

```c++
#include<bits/stdc++.h>
using namespace std;

typedef pair<int,int> PII;

const int N = 110;

int n,m;
int g[N][N]; // 存图
int d[N][N]; // 存的是每一个点到起点的距离
PII q[N * N];

int bfs(){ // 一般是有队列的，我们这里手动模拟队列，就不用stl啦
    int hh = 0, tt = 0;
    q[0] = {0, 0};
    
    // 将所有的距离初始化为-1，表示这个点我们没有走过
    memset(d, -1, sizeof d);
    d[0][0]=0;
    
    int dx[4] = {1,0,-1,0}, dy[4] = {0,1,0,-1};
    
    while(hh<=tt){
        auto t = q[hh++]; // 每次取出队头元素
        for(int i=0;i<4;i++){
            int x=t.first+dx[i], y=t.second+dy[i];
            if(x>=0 && x<n && y>=0 && y<m && g[x][y]==0 && d[x][y]==-1){
                d[x][y]=d[t.first][t.second] + 1;
                q[++tt]={x,y};
            }
        }
    }
    return d[n-1][m-1];
}

int main(){
    cin>>n>>m;
    for(int i=0;i<n;i++)
        for(int j=0;j<m;j++)
            cin>>g[i][j];
    cout<<bfs()<<endl;
}
```

```c++
#include<bits/stdc++.h>
using namespace std;

const int N = 100010, M = 2 * N;

int n;

// 存的链表头、每个结点的值、next指针
int h[N], e[M], ne[M], idx;
bool st[N];

int ans = N;

// 插入一条边a->b
void add(int a, int b){
    // 就是在a所对应的邻接表里插入一个结点b
    e[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}

// 返回以u为根的子树中点的数量
int dfs(int u){
    st[u] = true; // 标记一下，已经被搜过了
    int sum = 1, res = 0;
    // 遍历一下u的所有出边
    for(int i=h[u];i!=-1;i=ne[i]){
        int j = e[i];
        if(!st[j]){
            int s = dfs(j); // s表示当前子树的大小
            res = max(res,s);
            sum += s;
        }
    }
    res = max(n-sum,res);
    
    ans = min(ans,res);
    
    return sum;
}

int main(){
    cin>>n;
    memset(h, -1, sizeof h);
    
    for(int i=0;i<n-1;i++){
        int a,b;
        cin>>a>>b;
        add(a,b),add(b,a); // 本题是无向边
    }
    
    dfs(1);
    cout<<ans<<endl;
}

// res：删掉u后，所有连通块大小中的最大值
// sum：以u为根的子树大小

```

```c++
#include<bits/stdc++.h>
using namespace std;

int n,m;
const int N = 100010;
int h[N], e[N], ne[N], idx;
int d[N],q[N]; // d是距离，q是队列

void add(int a, int b){
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx++;
}

int bfs(){
    int hh = 0, tt = 0;
    q[0] = 1;
    
    memset(d, -1, sizeof d);
    d[1] = 0; // 最开始只有第1个点被遍历过啦，距离是0
    // 队列不空
    while(hh<=tt){
        int t = q[hh++];
        for(int i=h[t];i!=-1;i=ne[i]){
            int j = e[i];
            // 如果j没有被遍历过
            if(d[j] == -1){
                d[j] = d[t] + 1;
                q[++tt]=j;
            }
        }
    }
    return d[n];
}

int main(){
    memset(h, -1, sizeof h);
    cin>>n>>m;
    while(m--){
        int a,b;
        cin>>a>>b;
        add(a,b);
    }
    
    cout<<bfs()<<endl;
}
```

#### 有向图的拓扑序列

有向无环图又被称为拓扑图。

入度为0也就意味着没有其他点指向这个点，这个点一定是排在最前面的。

![[Pasted image 20260405193807.png]]

![[Pasted image 20260405193712.png|393]]

```c++
#include<bits/stdc++.h>
using namespace std;

const int N = 100010;

int n,m;
int h[N],e[N],ne[N],idx;
int q[N],d[N]; // 队列、入度

void add(int a,int b){
    e[idx]=b;
    ne[idx] = h[a];
    h[a] = idx++;
}

bool topsort(){
    int hh=0,tt=-1; // 为什么有时候tt定义为0 -1?
    for(int i=1;i<=n;i++){
        if(d[i]==0){
            q[++tt]=i;
        }
    }
    while(hh<=tt){
        int t = q[hh++];
        for(int i=h[t];i!=-1;i=ne[i]){
            int j=e[i];
            d[j]--;
            if(d[j]==0) q[++tt]=j;
        }
    }
    return tt==n-1;
}

int main(){
    cin>>n>>m;
    memset(h,-1,sizeof h);
    for(int i=0;i<m;i++){
        int a,b;
        cin>>a>>b;
        add(a,b);
        d[b]++;
    }
    if(topsort()){
        for(int i=0;i<n;i++) cout<<q[i]<<" ";
    }
    else cout<<"-1"<<endl;
    
}
```

## 4 最短路

### 朴素Dijkstra算法

适用于==稠密图==，而稠密图用==邻接矩阵==来存储。

![[Pasted image 20260406143002.png]]

![[Pasted image 20260406144119.png]]

==INF可表示为0x3f3f3f3f：==

![[Pasted image 20260406153858.png]]

```c++
#include<bits/stdc++.h>
using namespace std;

const int N = 510;
int n,m;
int g[N][N];
int dist[N]; // dist[i]=从起点1到i的当前最短距离
bool st[N]; // 表示每个点的最短路径是否已经确定了

int dijkstra(){
    memset(dist,0x3f,sizeof dist);
    dist[1]=0;
    // 循环n次，确定每个点的最短路径
    for(int i=0;i<n;i++){
        int t=-1;
        for(int j=1;j<=n;j++)
            // 在所有没确定的点中，找dist最小的那个点
            if(!st[j] && (t==-1 || dist[t]>dist[j]))
                t=j;
        st[t]=true;
        
        for(int j=1;j<=n;j++)
            dist[j]=min(dist[j],dist[t]+g[t][j]);
    }
    if(dist[n]==0x3f3f3f3f) return -1;
    return dist[n];
}

int main(){
    cin>>n>>m;
    memset(g,0x3f,sizeof g); // 初始化为INF
    while(m--){
        int a,b,c;
        cin>>a>>b>>c;
        g[a][b]=min(g[a][b],c); // 本题有重边和自环，咱们只用保留距离最短的那条边即可
    }
    int t = dijkstra();
    cout<<t<<endl;
}
```
