## 快速排序

###  快排

![[Pasted image 20260320104708.png|528]]

```c++
#include<bits/stdc++.h>
using namespace std;

const int N = 1e6 + 10;
int n;
int q[N];

void quick_sort(int q[],int l,int r){
    if(l>=r) return;//没有数或只有一个数，那就是排好序的，不用管了
    int x = q[(l+r)/2];//x作为分界点可以随便选取
    int i=l-1,j=r+1;//指针
    
    while(i<j){
        do i++; while(q[i]<x);//注意这里不能写<=
        do j--; while(q[j]>x);
        if(i<j) swap(q[i],q[j]);
    }
    quick_sort(q,l,j);
    quick_sort(q,j+1,r);
}

int main(){
    cin>>n;
    for(int k=0;k<n;k++){
        cin>>q[k];
    }
    quick_sort(q,0,n-1);
    for(int k=0;k<n;k++) cout<<q[k]<<" ";
    return 0;
}
```

### 归并排序

![[Pasted image 20260320130814.png|492]]

```c++
#include<bits/stdc++.h>
using namespace std;

int n;
const int N = 1e6 + 10;
int q[N],temp[N];

void merge_sort(int q[],int l,int r){
    if(l>=r) return;
    int mid = l+r >> 1;
    merge_sort(q,l,mid);
    merge_sort(q,mid+1,r);
    int i=l,j=mid+1;
    int k=0;
    while(i<=mid && j<=r){
        if(q[i]<=q[j]) temp[k++]=q[i++];
        else temp[k++]=q[j++];
    }
    while(i<=mid) temp[k++]=q[i++];
    while(j<=r) temp[k++]=q[j++];
    for(i=l,j=0;i<=r,j<k;i++,j++) q[i]=temp[j];
}

int main(){
    cin>>n;
    for(int i=0;i<n;i++) cin>>q[i];
    merge_sort(q,0,n-1);
    for(int i=0;i<n;i++) cout<<q[i]<<" ";
}
```

```c++
#include<bits/stdc++.h>
using namespace std;
int n,m;
const int N = 1e6 + 10;
int q[N];


int main(){
    cin>>n>>m;
    for(int i=0;i<n;i++) cin>>q[i];
    while(m--){
        int x;
        cin>>x;
        int l=0,r=n-1;
        //找左边界
        while(l<r){
            int mid=l+r >> 1;
            if(q[mid]>=x) r=mid;
            else l=mid+1;
        }
        if(q[l]!=x) cout<<"-1 -1"<<endl;
        else{
            cout<<l<<" ";
            //找右边界
            int l=0,r=n-1;
            while(l<r){
                int mid=l+r+1 >> 1;
                if(q[mid]<=x) l=mid;
                else r=mid-1;
            }
            cout<<l<<endl;
        }
    }
    return 0;
}
```

这道题其实坑挺多的，浮点数，并且要考虑负数的情况：

```c++
#include <bits/stdc++.h>
using namespace std;

int main(){
    double x;
    cin >> x;

    double l = -10000, r = 10000;//这里r不能等于x。因为x如果0~1的话，立方根其实是更大的（eg0.125的立方根为0.5）

    while(r-l > 1e-8){ // 这里-8是因为保留精度是6位 + 2 经验
        double mid = (l+r) / 2;
        if(mid*mid*mid >= x) r=mid;
        else l=mid;
    }

    cout << fixed << setprecision(6) << l << endl;

    return 0;
}
```
