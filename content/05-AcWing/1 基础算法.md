快排的分界点是整个数组的任意==值==，但为了避免边界出错情况，我们只记下面的模板，也就是
==x = q[(l + r) > 1]==

![[Pasted image 20260511143918.png|531]]

![[Pasted image 20260511150454.png|488]]

==快排模板==

```C
void quick_sort(int q[],int l,int r){
    if(l >= r) return;
    
    // 1 确定分界点
    int x = q[(l + r) >> 1];
    int i = l - 1, j = r + 1; // 指针我们写在边界的两侧
    while(i < j){
        do i++; while(q[i]<x);
        do j--; while(q[j]>x);
        if(i < j) swap(q[i],q[j]);
    }
    // 递归处理左右两边
    quick_sort(q,l,j);
    quick_sort(q,j+1,r);//为什么把这里的j换成i就是错？边界问题
}
```

归并的临界点是 ==mid=(l + r) > 1== ,这是个数组下标，是整个数组的中间位置的下标。

归并里最难的其实是将两个有序数组合二为一，用到双指针算法。（快排是划分最难~）

归并排序是稳定的，而快排是不稳定的。

## 归并排序

![[Pasted image 20260511164644.png]]

![[Pasted image 20260511165952.png]]

```C
#include<bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;

int n;
int q[N], tmp[N];

void merge_sort(int q[],int l,int r){
    if(l >= r) return;
    
    int mid = (l+r) >> 1;
    
    merge_sort(q,l,mid);
    merge_sort(q,mid+1,r);
    
    // 最麻烦的一步--归并
    int k = 0;// k表示当前tmp里已经有多少个数了，就是说两序列合并的时候已经合并了几个数了。
    int i=l,j=mid+1;//i指向左半边的起点，j指向右半边的起点
    while(i<=mid && j<=r){
        if(q[i]<=q[j]) tmp[k++]=q[i++];
        else tmp[k++]=q[j++];
    }
    // 此时有可能左半边没有循环完或者右半边没有循环完，我们把剩下的拿来
    while(i<=mid) tmp[k++]=q[i++];
    while(j<=r) tmp[k++]=q[j++];
    
    for(i=l,j=0;i<=r;i++,j++) q[i]=tmp[j];//
}

int main(){
    cin>>n;
    for(int i=0;i<n;i++) cin>>q[i];
    
    merge_sort(q,0,n-1);
    
    for(int i=0;i<n;i++) cout<<q[i]<<" ";
}
```

```C
#include<bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;
int n;
int q[N],tmp[N];
long long res;

void merge_sort(int q[],int l,int r){
    if(l>=r) return;
    
    int mid = (l+r) >> 1;
    int i=l,j=mid+1;
    
    merge_sort(q,l,mid);
    merge_sort(q,mid+1,r);
    
    int k=0;
    while(i<=mid && j<=r){
        if(q[i]<=q[j]) tmp[k++]=q[i++];
        else{
            res+=mid-i+1; //重点
            tmp[k++]=q[j++];
        }
    }
    while(i<=mid) tmp[k++]=q[i++];
    while(j<=r) tmp[k++]=q[j++];
    for(i=l,j=0;i<=r;i++,j++) q[i]=tmp[j];
}

int main(){
    cin>>n;
    for(int i=0;i<n;i++) cin>>q[i];
    
    merge_sort(q,0,n-1);
    cout<<res;
}
```

```C
#include<bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;

int n, m;
int q[N];

int main(){
    cin>>n>>m;
    for(int i=0;i<n;i++) cin>>q[i];
    
    while(m--){
        int x;
        cin>>x;
        
        int l=0,r=n-1;
        while(l<r){
            int mid=(l+r)>>1;
            
            if(q[mid]>=x) r=mid;
            else l=mid+1;
        }
        
        if(q[l]!=x) cout<<"-1 -1"<<endl;
        else{
            cout<<l<<" ";
            
            l=0,r=n-1;
            while(l<r){
                int mid=(l+r+1)>>1;
                
                if(q[mid]<=x) l=mid;
                else r=mid-1;
            }
            cout<<l<<endl;
        }
    }
}
```

## 切割木材

[蓝桥杯 切割木材](https://www.luogu.com.cn/problem/P16227)

这是我26年参加的蓝桥杯A组第四题吧，也是一道二分，我记不清自己是否写上去了hh

![[Pasted image 20260511202918.png|494]]

![[Pasted image 20260511202933.png|453]]

![[Pasted image 20260511203234.png|445]]

```C
#include<bits/stdc++.h>
using namespace std;

const int N = 1e6 + 10;

int n;
int k;
int a[N];

bool check(int x) {
    int cuts = 0;
    for(int i=0; i<n; i++) {
        cuts += (a[i] + x - 1) / x - 1;
        if(cuts > k) return false;
    }
    return true;
}

int main() {
    cin >> n >> k;
    int max_len = 0;
    for(int i=0; i<n; i++) {
        cin >> a[i];
        if(a[i] > max_len) max_len = a[i];
    }
    
    int l = 1, r = max_len;
    while(l < r) {
        int mid = (l + r) >> 1;
        if(check(mid)) r = mid;
        else l = mid + 1;
    }
    
    cout << l << endl;
    return 0;
}
```
